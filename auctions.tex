\begin{definition}[Bid-set protocol]\label{def:auction-framework}

    A \emph{bid-set protocol} defines the following parameters: the \emph{starting time} \tzero, and an integer $\Delta$ (looking ahead, this is a value that the real network delay is not expected to exceed). It also assumes a pre-appointed \emph{auctioneer}.
    Any party can be the auctioneer, but we assume w.l.o.g. a pre-appointed one.
    % We also define $\ttwo = \tzero + 2\Delta$.
    It exposes the following interface.
    \begin{itemize}
        \item function $\opSubmitBid{\bid}$: It is called by a bidder at round \tzero in order to submit a bid \bid.
        \oa{I think we can get rid of \tzero and write all results relative to some starting time.}
        \item event $\opResult{\bag, \ppcert}$: It is an event generated by the auctioneer. It contains a \emph{bid-set} \bag, which is a set of bids, and auxiliary information \ppcert.
        \oa{In the construction I will have the auctioneer, which is a ``special process'' generate this event. I do not do the final step that Dio described, where the auctioneer writes the result back to pod, because I am not sure how to model the protocol output in that case.}
    \end{itemize}
    It satisfies the following properties.
    \begin{enumerate}[leftmargin=*, align=left]
        \item Liveness:
        \begin{description}
            \item[Termination within \parAuctionTerm:]
                It the auctioneer is honest and the network is synchronous (that is, the network is delay is at most $\Delta$), $\opResult{\bag}$ is output by round $X$.

                \oa{Notice that in asynchrony, an honest auctioneer may be blamed for liveness failure.}
                \hue{Liveness: In a synchronous network, the auction by an honest auctioneer concludes by $t_0 + 2\Delta + 2\delta$.}
            \end{description}
        \item Safety:
        
        \begin{description}
            \item [Accountable censorship resistance:]
            For a \opSubmitBid{\bid} called by an honest bidder in a synchronous network, either $\bid \in \bag$ or the auctioneer is accountable.
            Moreover, \emph{fair punishment} holds: an honest auctioneer cannot be held accountable for censoring bids, even in asynchronous networks.  \oa{Notice that, even in asynchrony, an honest auctioneer cannot be blamed, in the construction we present.}
            \hue{Bid inclusion: In a synchronous network, all honest bids are included}

\subsection{Construction}

\begin{construction}\label{const:bidset}
    We show the construction in Algorithms~\ref{alg:bid-set-bidder} and ~\ref{alg:bid-set-auctioneer}.
\end{construction}

\begin{theorem}[Bid-set security]
    \Cref{const:bidset}, when instantiated with a secure pod protocol, is a secure bid-set protocol.
\end{theorem}


\begin{algorithm}
    \caption{Our bid-set construction. It makes use of a pod instance \podInst and is parametrized by $\tzero$ and $\Delta$. Code for a bidder, run at time \tzero.}
    \label{alg:bid-set-bidder}
    \begin{algorithmic}[1]
        \Function{\opSubmitBid{\bid}}{}
            \State{\podInst.\opwrite{\bid}}
        \EndFunction 
\end{algorithmic}
\end{algorithm}

\begin{algorithm}
    \caption{Our bid-set construction. It makes use of a pod instance \podInst and is parametrized by $\tzero$ and $\Delta$. Code for an auctioneer, run at time $\ttwo = \tzero + 2 \Delta$.}
    \label{alg:bid-set-auctioneer}
    \begin{algorithmic}[1]
        \Function{\opRunAuction{}}{}
            \Let{(\txSet, \rperf, \ppcert)}{\podInst.\opreadall{}}
            \While{\rperf < \ttwo}\label{line:bidset-wait-pp}
                \Let{(\txSet, \rperf, \ppcert)}{\podInst.\opreadall{}}
            \EndWhile{}
            \Let{\bag}{\{\tx \mid (\tx, \cdot, \cdot, \cdot, \cdot) \in \txSet\}}
            \State{\textbf{output event} \opResult{\bag, \ppcert}}
        \EndFunction
\end{algorithmic}
\end{algorithm}

\begin{algorithm}
    \caption{The \identifyAuctioneer{} function for \Cref{const:bidset}, instantiated with an instance of \Cref{const:pod-core} as \podInst, run by a set of replicas $\allreps = \{\rep_1, \cdots, \rep_n\}$ with public keys $\{\pk1, \cdots, \pk_n \}$, and using the \opMed{} operation defined by \Cref{const:pod-core}.}
    \label{alg:identify-bidset}
    \begin{algorithmic}[1]
        \Function{\identify{\txcert, (\bag, \ppcert)}}{}
            \Let{\txVotesPerf}{[ \, ]}
            \For{$\vote \in \ppcert$} \label{line:bidset-identify-votes-begin}
                \Let{(\tx, \tsp, \sig, \rep_j)}{\vote}
                \Let{\txVotesPerf}{\txVotesPerf \concat \tsp}
                \If{\opVerify{$\pk_j, (\tx, \tsp), \sig$} = 0 }
                    \State{\textbf{return} $\true$}
                \EndIf{}
            \EndFor{}\label{line:bidset-identify-votes-end}
            
            \State{\text{sort \txVotesPerf in increasing order}} \label{line:bidset-identify-rperf-begin}
            \Let{\txVotesPerf}{ [ 0, \stackrel{\beta \text{ times}}{\ldots}, 0 ] \concat \txVotesPerf}
            \Let{\rperf}{\opMed{$\txVotesPerf[:\alpha]$}}
            \If{$\rperf \leq \ttwo$}
                \State{\textbf{return} $\true$}
            \EndIf \label{line:bidset-identify-rperf-end}
            \State{\wip{Verify validity of \txcert: contains \alpha votes, check all sigs, check that votes are on the same \tx}} \label{line:bidset-identify-inclusion-begin}
            \Let{\tx}{\text{\wip{the common transaction in all votes in \txcert}}}
            \Let{\rconf}{\text{\wip{the median of all votes in \txcert}}}
            \If{$\rconf \leq \ttwo \textbf{ and } \tx \not \in \bag$}
                \State{\textbf{return} $\true$}
            \EndIf \label{line:bidset-identify-inclusion-end}
        \EndFunction
    \end{algorithmic}
    \end{algorithm}

\begin{lemma}
    \Cref{const:bidset}, instantiated with a secure pod protocol that satisfies the \emph{past-perfection within $\parPerf$} property for $\parPerf = \delta$, satisfies the \emph{Termination within \parAuctionTerm} property for $\parAuctionTerm = \ttwo + \delta$.
\end{lemma}
\begin{proof}
    The auctioneer starts running \opRunAuction{} at time \ttwo. From the \emph{past-perfection within $\parPerf$} property of pod, for $\parPerf = \delta$, the condition of line~\ref{line:bidset-wait-pp} becomes true by round $\ttwo + \delta$.
\end{proof}

\begin{lemma}
    \Cref{const:bidset}, instantiated with a $\parTemp-timely$ pod protocol for $\parTemp=\delta$, satisfies the \emph{accountable censorship resistance} property.
\end{lemma}
\begin{proof}   
    Assume the auctioneer is honest, and an honest bidder calls \opSubmitBid{\bid} at time \tzero, and the network is synchronous, i.e., the $\Delta$ used in \Cref{const:bidset} and the actual network delay $\delta$ satisfy $\delta \leq \Delta$. We will show that $\bid \in \bag$.
    First, the pod view $\podView{\auc}{\round}$ of the auctioneer \auc on the round \round when it constructs \bag satisfies $\podView{\auc}{\round}.\rperf > \ttwo$. 
    Second, from the \emph{confirmation within \parConf} property of pod, the transaction containing \bid becomes confirmed, and from the \emph{\parTemp-timeliness} property of pod, it gets a confirmation round $\rconf \leq \tzero + \parTemp$. For $\parTemp = \delta$, and since $\delta \leq \Delta$, we get that $\rconf \leq \ttwo$.
    Hence, from the past-perfection safety property of pod we get that $\bid \in \podView{\auc}{\round}$, and, since the auctioneer is honest, $\bid \in \bag$.
    
    For the accountability part, assume an honest bidder calls \opSubmitBid{\bid} at time \tzero, the network is synchronous, and \bid is censored. 
    Following the same reasoning as above, the transaction \tx containing \bid becomes confirmed with $\rconf \leq \ttwo$, and any honest party can observe $(\tx, \rconf, \cdot, \cdot, \txcert)$ in their view of the pod, where \txcert is the auxiliary data associated by \tx, as returned by pod.
    We will show that how the auctioneer can be made accountable, assuming that \podInst is an instance of \Cref{const:pod-core}, and that \Cref{const:pod-core} is a secure pod protocol.
    Following \Cref{sec:accountable-safety}, we show an \identifyAuctioneer{\txcert, (\bag, \ppcert)} function (\Cref{alg:identify-bidset}), that, on input \txcert and the bag \bag and auxiliary data \ppcert, returned by the auctioneer in \opResult{\bag, \ppcert}, outputs $\true$ when safety is violated due to misbehavior of the auctioneer (i.e., it identifies the auctioneer as malicious), and $\false$ if the auctioneer is honest.
    \Cref{alg:identify-bidset} performs the following checks:
    
    \emph{Check 1}: Verify whether the votes that the auctioneer has included in \ppcert are valid votes, obtained from the replicas that run the instance \podInst (lines~\cref{line:bidset-identify-votes-begin}-\cref{line:bidset-identify-votes-end}). If this is not the case, the auctioneer has misbehaved.

    \emph{Check 2}: Compute the \rperf from the timestamps found in the votes in \ppcert (lines~\ref{line:bidset-identify-rperf-begin}-\ref{line:bidset-identify-rperf-end}). This \rperf must be at least \ttwo.
    
    \emph{Check 3}: Verify whether \txcert contains \alpha valid votes on the same transaction \tx, produced by the replicas running \podInst, and compute \rconf as the median of the timestamps in \txcert.
    
    Finally, we show that an honest auctioneer cannot be framed.
    If the auctioneer has followed \Cref{alg:bid-set-auctioneer}, then \ppcert will contain valid votes, as output by the \opreadall{} function of \Cref{const:pod-core}, hence \emph{Check 1} will pass.
    Moreover, an honest auctioneer waits until the past-perfect round returned by the pod is larger than \ttwo, hence \emph{Check 2} will pass.
    Regarding \emph{Check 3}, observe that
    % any \txcert that is deemed valid by \identifyAuctioneer{} contains \alpha votes, hence at least $\alpha-\beta$ votes from valid replicas. Moreover,
    for \identifyAuctioneer{} to compute $\rconf \leq \ttwo$, \txcert must contain at least $\lfloor \alpha/2 \rfloor + 1$ votes on \tx with a timestamp smaller or equal than \ttwo, and at least $\lfloor \alpha/2 \rfloor + 1 - \beta$ of them must be from honest replicas. Call this set $\allreps'$.
    The honest auctioneer, in order to output a past-perfect round greater than \ttwo, must have received timestamps greater than \ttwo by at least $n - \lfloor a/2 \rfloor + \beta$ replicas (from \Cref{lem:assign-min-max}).
    By counting arguments, at least one of these timestamps must be from one of the honest replicas in $\allreps'$, and, since honest replicas do not omit transactions, that replica will have sent a vote on \tx to the auctioneer. Hence, the auctioneer will include that \tx in \bag.
    \wip{Can this be derived directly from the past-perfection property of pod? }
\end{proof}
